<script>
const $=s=>document.querySelector(s), $$=s=>[...document.querySelectorAll(s)];
const esc=s=>String(s??'').replace(/[&<>'"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'}[c]));
let evolutionEras=[];

function switchView(name){
  $('#overviewView').hidden=name!=='overview';
  $('#timelineView').hidden=name!=='timeline';
  $('#trendView').hidden=name!=='trend';
  $('#trajectoryView').hidden=name!=='trajectory';
  $$('[data-view]').forEach(b=>b.classList.toggle('active',b.dataset.view===name));
  const hash=name==='overview'?'':'#'+name;
  history.replaceState(null,'','evolution.html'+hash);
}
$$('[data-view]').forEach(b=>b.addEventListener('click',()=>switchView(b.dataset.view)));
if(location.hash==='#timeline')switchView('timeline'); else if(location.hash==='#trend')switchView('trend'); else if(location.hash==='#trajectory')switchView('trajectory');

function renderEraTimeline(eras){
  const host=$('#eraTimeline');
  host.innerHTML=eras.length?eras.map(e=>`<article class="era ${e.status==='current'?'current':''}"><div class="era-head"><strong>${esc(e.display_label||e.period)}</strong><small>${esc(e.start_date||'—')} ～ ${esc(e.end_date||'現在')}</small></div><p>${esc(e.description||'')}</p></article>`).join(''):'<div class="empty">尚無時期資料。</div>';
}

function renderTimelineEvents(){
  const host=$('#timelineEventList'); if(!host)return;
  const rows=[...((window.LOC_EVENT_DATA&&window.LOC_EVENT_DATA.events)||[])].sort((a,b)=>String(b.date||'').localeCompare(String(a.date||'')));
  host.innerHTML=rows.length?rows.map(e=>`<article class="event-card"><div class="event-head"><div><strong>${esc(e.event_title||e.title||'未命名事件')}</strong><small>${esc(e.date||'')} · ${esc(e.event_type||'')}${e.era_label?' · '+esc(e.era_label):''}</small></div><small>${esc(e.status||'recorded')}</small></div>${e.description?`<p>${esc(e.description)}</p>`:''}${e.state_after?`<p><strong>State →</strong> ${esc(e.state_after)}</p>`:''}${e.legacy_era?`<p class="source-note">歷史標記：${esc(e.legacy_era)}；現行顯示依日期映射。</p>`:''}</article>`).join(''):'<div class="empty">尚無事件資料。</div>';
}

function sourceRows(source,period){
  const data=window.LOC_PERIOD_ANALYSIS_DATA||{};
  return (data[source]||[]).filter(x=>x.canonical_period===period);
}
function sourceCard(label,row){
  const weight=row.work_count??row.document_count??0;
  return `<div class="trajectory-box"><b>${esc(label)} · ${esc(row.period)}</b><small>${weight?`樣本 ${esc(weight)}`:'歷史切片'}</small><div class="chips">${(row.keywords||[]).map(k=>`<span class="chip">${esc(k.term)} ${esc(k.percent)}%</span>`).join('')}</div></div>`;
}
function renderTrend(eras){
  const host=$('#trendGrid'); if(!host)return;
  host.innerHTML=eras.map(era=>{
    const music=sourceRows('loc3',era.period),text=sourceRows('loc6',era.period);
    const blocks=[...music.map(x=>sourceCard('LOC3',x)),...text.map(x=>sourceCard('LOC6',x))].join('');
    return `<article class="keyword-card"><strong>${esc(era.display_label||era.period)}</strong><small>${esc(era.start_date||'—')} ～ ${esc(era.end_date||'現在')}</small>${blocks||'<div class="chips"><span class="chip">尚無分析資料</span></div>'}</article>`;
  }).join('');
}

function weightedVector(rows){
  const acc=new Map(),den=new Map();
  for(const row of rows){
    const w=Number(row.work_count??row.document_count??1)||1;
    for(const k of row.keywords||[]){
      const term=String(k.term||''); if(!term)continue;
      acc.set(term,(acc.get(term)||0)+Number(k.percent||0)*w);
      den.set(term,(den.get(term)||0)+w);
    }
  }
  return new Map([...acc].map(([term,sum])=>[term,Number((sum/(den.get(term)||1)).toFixed(1))]));
}
function eraVector(period){
  const music=sourceRows('loc3',period);
  if(music.length)return weightedVector(music);
  return weightedVector(sourceRows('loc6',period));
}
function delta(from,to){
  const terms=new Set([...from.keys(),...to.keys()]);
  const rows=[...terms].map(term=>({term,delta:Number(((to.get(term)||0)-(from.get(term)||0)).toFixed(1))}));
  return {up:rows.filter(x=>x.delta>0).sort((a,b)=>b.delta-a.delta).slice(0,4),down:rows.filter(x=>x.delta<0).sort((a,b)=>a.delta-b.delta).slice(0,4)};
}
function renderTrajectory(eras){
  const host=$('#trajectoryList'); if(!host)return;
  const cards=[];
  for(let i=1;i<eras.length;i++){
    const fromEra=eras[i-1],toEra=eras[i],a=eraVector(fromEra.period),b=eraVector(toEra.period);
    if(!a.size||!b.size)continue;
    const d=delta(a,b),chip=(x,up)=>`<span class="chip">${esc(x.term)} ${up?'+':''}${esc(x.delta)}pt</span>`;
    cards.push(`<article class="trajectory-card"><div class="trajectory-head"><strong>${esc(fromEra.display_label||fromEra.period)} → ${esc(toEra.display_label||toEra.period)}</strong><small>${esc(toEra.start_date||'—')}</small></div><div class="trajectory-delta"><div class="trajectory-box"><b>比重上升</b><div class="chips">${d.up.length?d.up.map(x=>chip(x,true)).join(''):'<span class="chip">無明顯上升</span>'}</div></div><div class="trajectory-box"><b>比重下降</b><div class="chips">${d.down.length?d.down.map(x=>chip(x,false)).join(''):'<span class="chip">無明顯下降</span>'}</div></div></div><div class="trajectory-note">${esc(toEra.description||'')}</div></article>`);
  }
  host.innerHTML=cards.join('')||'<div class="empty">目前沒有足夠的相鄰時期分析資料可形成軌跡。</div>';
}

window.LOCPeriods.load({force:true}).then(periodData=>{
  const eras=[...(periodData.eras||[])].sort((a,b)=>Number(a.order||0)-Number(b.order||0));
  evolutionEras=eras;
  const current=eras.find(e=>e.status==='current')||eras[eras.length-1];
  $('#eraCount').textContent=eras.length||'0';
  $('#eraCurrent').textContent=current?.period||'—';
  $('#eraStart').textContent=current?.start_date||'—';
  renderEraTimeline(eras);renderTimelineEvents();renderTrend(eras);renderTrajectory(eras);
}).catch(err=>{
  $('#eraTimeline').innerHTML='<div class="empty">靜態資料載入失敗：'+esc(err.message)+'</div>';
  $('#trendGrid').innerHTML='<div class="empty">靜態資料載入失敗。</div>';
  $('#trajectoryList').innerHTML='<div class="empty">軌跡資料載入失敗。</div>';
});
</script>
